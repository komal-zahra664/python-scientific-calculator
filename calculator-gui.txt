def insert_value(value):
    current = expression_var.get()

    if current == "Error":
        current = ""

    expression_var.set(current + value)


def clear():
    expression_var.set("")


def backspace():
    current = expression_var.get()
    expression_var.set(current[:-1])


def calculate():
    expression = expression_var.get()

    if not expression:
        return

    try:
        formatted_expression = expression.replace("^", "**")

        allowed_values = {
            "sin": math.sin,
            "cos": math.cos,
            "tan": math.tan,
            "sqrt": math.sqrt,
            "log": math.log10,
            "ln": math.log,
            "pi": math.pi,
            "e": math.e,
            "abs": abs
        }

        result = eval(
            formatted_expression,
            {"__builtins__": {}},
            allowed_values
        )

        if isinstance(result, float):
            result = round(result, 10)

        history_text = f"{expression} = {result}"

        history_items.insert(0, history_text)

        history_list.insert(0, history_text)

        expression_var.set(str(result))

    except:
        expression_var.set("Error")


def percentage():
    current = expression_var.get()

    if current:
        expression_var.set(f"({current})/100")


def square():
    current = expression_var.get()

    if current:
        expression_var.set(f"({current})^2")


def square_root():
    current = expression_var.get()

    if current:
        expression_var.set(f"sqrt({current})")
    else:
        expression_var.set("sqrt(")


def toggle_sign():
    current = expression_var.get()

    if not current:
        return

    try:
        if current.startswith("-"):
            expression_var.set(current[1:])
        else:
            expression_var.set("-" + current)
    except:
        pass


def clear_history():
    history_items.clear()
    history_list.delete(0, tk.END)

button_widgets = []


def create_button(
    text,
    row,
    column,
    command,
    button_type="number",
    columnspan=1
):

    button = tk.Button(
        buttons_frame,
        text=text,
        font=("Segoe UI", 15, "bold"),
        bd=0,
        relief="flat",
        cursor="hand2",
        command=command
    )

    button.grid(
        row=row,
        column=column,
        columnspan=columnspan,
        sticky="nsew",
        padx=5,
        pady=5
    )

    button.button_type = button_type

    button.bind(
        "<Enter>",
        lambda event, btn=button: button_hover(btn, True)
    )

    button.bind(
        "<Leave>",
        lambda event, btn=button: button_hover(btn, False)
    )

    button_widgets.append(button)

    return button

# =========================
# HOVER EFFECT
# =========================

def button_hover(button, hovering):

    theme = themes[current_theme]

    button_type = button.button_type

    if button_type == "number":

        color = (
            theme["number_hover"]
            if hovering
            else theme["number"]
        )

    elif button_type == "operator":

        color = (
            theme["operator_hover"]
            if hovering
            else theme["operator"]
        )

    elif button_type == "equal":

        color = (
            theme["equal_hover"]
            if hovering
            else theme["equal"]
        )

    elif button_type == "danger":

        color = (
            theme["danger_hover"]
            if hovering
            else theme["danger"]
        )

    else:

        color = (
            theme["special_hover"]
            if hovering
            else theme["special"]
        )

    button.configure(bg=color)

def apply_theme():

    theme = themes[current_theme]

    window.configure(
        bg=theme["bg"]
    )

    main_frame.configure(
        bg=theme["bg"]
    )

    calculator_frame.configure(
        bg=theme["bg"]
    )

    header_frame.configure(
        bg=theme["bg"]
    )

    buttons_frame.configure(
        bg=theme["bg"]
    )

    display_frame.configure(
        bg=theme["card"]
    )

    history_frame.configure(
        bg=theme["card"]
    )

    title.configure(
        bg=theme["bg"],
        fg=theme["text"]
    )

    history_title.configure(
        bg=theme["card"],
        fg=theme["text"]
    )

    display.configure(
        bg=theme["display"],
        fg=theme["text"],
        insertbackground=theme["text"]
    )

    history_list.configure(
        bg=theme["display"],
        fg=theme["text"],
        selectbackground=theme["operator"],
        selectforeground="white"
    )

    clear_history_button.configure(
        bg=theme["danger"],
        fg="white",
        activebackground=theme["danger_hover"],
        activeforeground="white"
    )

    for button in button_widgets:

        button_type = button.button_type

        if button_type == "number":

            button.configure(
                bg=theme["number"],
                fg=theme["text"],
                activebackground=theme["number_hover"]
            )

        elif button_type == "operator":

            button.configure(
                bg=theme["operator"],
                fg="white"
            )

        elif button_type == "equal":

            button.configure(
                bg=theme["equal"],
                fg="white"
            )

        elif button_type == "danger":

            button.configure(
                bg=theme["danger"],
                fg="white"
            )

        else:

            button.configure(
                bg=theme["special"],
                fg="white"
            )


def toggle_theme():

    global current_theme

    if current_theme == "dark":

        current_theme = "light"

        theme_button.configure(
            text="🌙 Dark"
        )

    else:

        current_theme = "dark"

        theme_button.configure(
            text="☀ Light"
        )

    apply_theme()


theme_button.configure(
    command=toggle_theme
)
